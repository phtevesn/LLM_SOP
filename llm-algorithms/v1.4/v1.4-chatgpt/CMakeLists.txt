cmake_minimum_required(VERSION 3.13)
project(sop_exact VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(SOP_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
add_library(sop_core src/instance.cpp src/assignment.cpp src/history.cpp src/heuristic.cpp src/topology.cpp src/solver.cpp)
target_include_directories(sop_core PUBLIC include)
target_link_libraries(sop_core PUBLIC Threads::Threads)
target_compile_options(sop_core PRIVATE -Wall -Wextra -Wpedantic)
if(SOP_SANITIZE)
  target_compile_options(sop_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_libraries(sop_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(sop_solver src/main.cpp src/cli.cpp)
target_link_libraries(sop_solver PRIVATE sop_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_assignment tests/test_assignment.cpp)
  target_link_libraries(test_assignment PRIVATE sop_core)
  add_test(NAME assignment_oracle COMMAND test_assignment)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME solver_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_solver.py $<TARGET_FILE:sop_solver>)
  endif()
endif()
