cmake_minimum_required(VERSION 3.16)
project(sop_exact VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(SOP_NATIVE "Optimize for the compiling machine's CPU" OFF)
option(SOP_BUILD_TESTS "Build independent correctness tests" ON)
set(SOP_SANITIZERS "" CACHE STRING "Comma-separated GCC/Clang sanitizers")
add_library(sop_core src/common.cpp src/instance.cpp src/assignment.cpp src/heuristics.cpp
    src/history.cpp src/dominance.cpp src/tail_dp.cpp src/parallel.cpp src/solver.cpp src/cli.cpp)
target_include_directories(sop_core PUBLIC include)
target_link_libraries(sop_core PUBLIC Threads::Threads)
target_compile_options(sop_core PRIVATE -Wall -Wextra -Wpedantic)
if(SOP_NATIVE)
    target_compile_options(sop_core PRIVATE -march=native)
endif()
if(SOP_SANITIZERS)
    target_compile_options(sop_core PUBLIC -fsanitize=${SOP_SANITIZERS} -fno-omit-frame-pointer -g)
    target_link_options(sop_core PUBLIC -fsanitize=${SOP_SANITIZERS})
endif()
add_executable(sop_exact src/main.cpp)
target_link_libraries(sop_exact PRIVATE sop_core)
install(TARGETS sop_exact RUNTIME DESTINATION bin)
if(SOP_BUILD_TESTS)
    enable_testing()
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE sop_core)
    add_test(NAME core COMMAND core_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py
            --solver $<TARGET_FILE:sop_exact>)
    endif()
endif()
